param(
    [string]$Base = "http://localhost:8099"
)

# ============================================================================
#  Hospital HC - pruebas del subsistema de acceso y roles
#
#  Uso (con el servidor ya arrancado y una base de datos recien creada):
#      1. Deten el servidor y borra la carpeta data\
#      2. java -jar hospital.jar 8099
#      3. powershell -ExecutionPolicy Bypass -File test\TestAccesoRoles.ps1
#
#  El script usa usuarios desechables con sufijo aleatorio para el bloqueo y
#  el cambio de contrasena, de modo que las pruebas dontienen cuentas ajenas.
#
#  Comprueba: proteccion de rutas, permisos de los cuatro roles, cabeceras de
#  seguridad, CSRF, bloqueo por fuerza bruta, cierre de sesion, cambio de
#  contrasena y administracion de usuarios.
# ============================================================================

$ErrorActionPreference = 'Continue'
$global:pass = 0
$global:fail = 0

function Check($nombre, $condicion, $detalle = "") {
    if ($condicion) {
        $script:pass++
        Write-Host ("  [OK]    " + $nombre) -ForegroundColor Green
    } else {
        $script:fail++
        Write-Host ("  [FALLA] " + $nombre + "   " + $detalle) -ForegroundColor Red
    }
}

# --- Cliente HTTP minimo: no lanza excepcion ante 4xx/5xx y no sigue redirecciones ---

function Invoke-Http {
    param(
        [string]$Metodo = 'GET',
        [string]$Ruta = '/',
        $Cookies = $null,
        [hashtable]$Form = $null,
        [string]$Origen = ''
    )

    $peticion = [System.Net.HttpWebRequest]::Create("$Base$Ruta")
    $peticion.Method = $Metodo
    $peticion.AllowAutoRedirect = $false
    $peticion.UserAgent = 'HospitalHC-Pruebas/1.0'
    $peticion.Timeout = 15000
    $peticion.ReadWriteTimeout = 15000
    $peticion.KeepAlive = $false

    if ($null -ne $Cookies) { $peticion.CookieContainer = $Cookies }
    if ($Origen -ne '') { $peticion.Headers.Add('Origin', $Origen) }

    if ($null -ne $Form) {
        $pares = @()
        foreach ($clave in $Form.Keys) {
            $pares += ([uri]::EscapeDataString([string]$clave) + '=' +
                       [uri]::EscapeDataString([string]$Form[$clave]))
        }
        $cuerpo = [System.Text.Encoding]::UTF8.GetBytes(($pares -join '&'))
        $peticion.ContentType = 'application/x-www-form-urlencoded; charset=UTF-8'
        $peticion.ContentLength = $cuerpo.Length
        $flujo = $peticion.GetRequestStream()
        $flujo.Write($cuerpo, 0, $cuerpo.Length)
        $flujo.Close()
    }

    $respuesta = $null
    $estado = 0
    $cuerpoTexto = ''
    $ubicacion = ''
    $cabeceraCookie = ''
    try {
        $respuesta = $peticion.GetResponse()
    } catch [System.Net.WebException] {
        $respuesta = $_.Exception.Response
    }

    if ($null -ne $respuesta) {
        $estado = [int]$respuesta.StatusCode
        $ubicacion = $respuesta.Headers['Location']
        $cabeceraCookie = $respuesta.Headers['Set-Cookie']
        $lector = New-Object System.IO.StreamReader($respuesta.GetResponseStream())
        $cuerpoTexto = $lector.ReadToEnd()
        $lector.Close()
        $respuesta.Close()
    }

    return [pscustomobject]@{
        Codigo = $estado
        Location = $ubicacion
        Html = $cuerpoTexto
        SetCookie = $cabeceraCookie
    }
}

function New-Cookies { return (New-Object System.Net.CookieContainer) }

function Get-Pagina($ruta, $cookies) { return Invoke-Http -Metodo GET -Ruta $ruta -Cookies $cookies }
function Post-Form($ruta, $form, $cookies) { return Invoke-Http -Metodo POST -Ruta $ruta -Form $form -Cookies $cookies -Origen $Base }

function Entrar($usuario, $clave) {
    $c = New-Cookies
    $r = Invoke-Http -Metodo POST -Ruta '/login' -Form @{ username = $usuario; password = $clave } -Cookies $c -Origen $Base
    if ($r.Codigo -ne 302) { Write-Host ("    (no se pudo iniciar sesion como " + $usuario + ")") -ForegroundColor Yellow }
    return $c
}

# Extrae el identificador de la fila que corresponde a un usuario concreto,
# en vez de emparejar el enlace con el nombre equivocado.
function Obtener-IdUsuario($html, $usuario) {
    foreach ($fila in ($html -split '<tr>')) {
        if ($fila -notmatch [regex]::Escape($usuario)) { continue }
        if ($fila -match '/usuarios/(\d+)/editar') { return $Matches[1] }
    }
    return $null
}
